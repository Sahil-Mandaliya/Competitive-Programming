vector<int> findNextPermutation(vector<int> v) {
    int i,j,n,x;
    n=v.size();
    vector<int> nxt=v;
    
    int stop=-1;
    for(i=n-2;i>=0;i--) { // O(n)
        if(v[i] < v[i+1]) {
            stop=i;
            break;
        }
    }
    // cout<<stop<<endl;
    if(stop==-1) {
        return v;
    }
    
    for(i=n-1;i>stop;i--) {  // O(n)
        if (nxt[i] > nxt[stop]) {
            x=nxt[stop];
            nxt[stop]=nxt[i];
            nxt[i]=x;
            break;
        }
    }

    i=stop+1;
    j=n-1;
    while(i<=j) {. // O(n)
        x=nxt[j];
        nxt[j]=nxt[i];
        nxt[i]=x;
        i++;
        j--;
    }

    return nxt;
}

void printAns(vector<int> v) {
    int n=v.size();
    for(int i=0;i<n;i++) {
        cout<<v[i]<<" ";
    }
    cout<<endl;
}

int countPermutations(int n) {
    int total=1;
    for(int i=1;i<=n;i++) {
        total*=i;
    }
    return total;
}

void solve(vector<int> v, int n, int k) {
    // int n=v.size();
    int i;
    vector<vector<int>> p;
    vector<int> prev=v;
    p.push_back(v);
    int total=countPermutations(n);
    for(i=0;i<total;i++) { // O(n!)
        printAns(prev);
        vector<int> nextPermutation = findNextPermutation(prev);
        prev=nextPermutation;
    }
   

}
